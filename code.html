async function goiConTim(em) {
const chanThanh = gather(20);
const moi = await doi(chanThanh);
if (!em.dongY) return cho();
const hoa = await em.trao("hoa");
const qua = goi(conTim, hoa);
while (hoa.tanUa) {
qua.nhipDap += tick();
}
await em.veDay();
const tay = giu(em, chatHon);
if (cachRoi) return khong(tay);
return nangNiu(em, suotDoi);
}
